    async get_power_state(self, hostname: str) -> str: resolve BMC host, GET https://{bmc}/redfish/v1/Systems/{system_id}, raise_for_status, return response JSON "PowerState". Wrap httpx.HTTPError in RedfishError using extract_error_message. Log with structlog (hostname, bmc_host — never credentials).

    async power_action(self, hostname: str, action: str, *, timeout: float | None = None) -> str: validate action is in _ACTION_TARGET_STATE (raise RedfishError if not). Look up target state. Call get_power_state. If current == target, log and return current (D-03 silent success). Otherwise call _post_reset, then _poll_power_state with timeout (default self._poll_timeout).

    async _post_reset(self, hostname: str, action: str) -> None: resolve BMC host, POST to https://{bmc}/redfish/v1/Systems/{system_id}/Actions/ComputerSystem.Reset with JSON body {"ResetType": action}. Wrap errors in RedfishError.

    async _poll_power_state(self, hostname: str, target: str, timeout: float) -> str: loop with asyncio.get_event_loop().time() deadline, call get_power_state, return if matches target, await asyncio.sleep(self._poll_interval). On timeout raise RedfishError with message including target state and timeout value.
